-- ============================================================================
-- TABLE 1: dim_emoji (Sentiment Polarity & Crowd Energy Lookup)
-- ============================================================================
CREATE TABLE dim_emoji (
    emoji_id           VARCHAR(10) PRIMARY KEY,
    emoji_symbol       VARCHAR(10) NOT NULL,
    emoji_name         VARCHAR(30) NOT NULL,
    mood_category      VARCHAR(20) NOT NULL,    -- 'Euphoria', 'Appreciation', 'Shock', 'Amusement', 'Heartbreak', 'Frustration', 'Hype'
    sentiment_polarity NUMERIC(4,2) NOT NULL,   -- Range: -1.00 (Extreme Negative) to +1.00 (Extreme Positive)
    arousal_intensity  NUMERIC(4,2) NOT NULL,   -- Range:  0.10 (Calm) to 1.00 (Explosive Crowd Frenzy)
    is_positive        BOOLEAN NOT NULL,
    updated_at         TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- ============================================================================
-- TABLE 2: dim_player (IPL 2026 Final Playing XI Squads: MI & CSK)
-- ============================================================================
CREATE TABLE dim_player (
    player_id           VARCHAR(15) PRIMARY KEY,
    player_name         VARCHAR(50) NOT NULL,
    jersey_number       INT NOT NULL,
    team_code           VARCHAR(10) NOT NULL,    -- 'MI' or 'CSK'
    team_full_name      VARCHAR(40) NOT NULL,
    player_country      VARCHAR(30) NOT NULL,
    player_role         VARCHAR(20) NOT NULL,    -- 'Batter', 'Bowler', 'All-Rounder', 'WK-Batter'
    batting_style       VARCHAR(25) NOT NULL,
    bowling_style       VARCHAR(35) NOT NULL,
    fan_popularity_tier VARCHAR(20) NOT NULL,    -- 'TIER_1_ICON', 'TIER_2_STAR', 'TIER_3_PRO'
    is_captain          BOOLEAN NOT NULL DEFAULT FALSE,
    updated_at          TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);
